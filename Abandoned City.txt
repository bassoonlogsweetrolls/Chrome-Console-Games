function boxParts(w, h, d, tw, th) {
  const g = new THREE.BoxGeometry(w, h, d), uv = g.attributes.uv;
  const sc = [[d / tw, h / th], [d / tw, h / th], [w / 8, d / 8], [w / 8, d / 8], [w / tw, h / th], [w / tw, h / th]];
  for (let f = 0; f < 6; f++) for (let i = 0; i < 4; i++) { const k = f * 4 + i; uv.setXY(k, uv.getX(k) * sc[f][0], uv.getY(k) * sc[f][1]); }
  const idx = Array.from(g.index.array), pick = fs => fs.flatMap(f => idx.slice(f * 6, f * 6 + 6));
  const s = g.clone(), t = g.clone(); s.setIndex(pick([0, 1, 4, 5])); t.setIndex(pick([2, 3]));
  g.dispose(); return [s, t];
}
function mergeGeos(list) {
  let vc = 0, ic = 0;
  for (const g of list) { vc += g.attributes.position.count; ic += g.index.count; }
  const pos = new Float32Array(vc * 3), nor = new Float32Array(vc * 3), uv = new Float32Array(vc * 2), idx = new Uint32Array(ic);
  let vo = 0, io = 0;
  for (const g of list) {
    pos.set(g.attributes.position.array, vo * 3); nor.set(g.attributes.normal.array, vo * 3); uv.set(g.attributes.uv.array, vo * 2);
    const ia = g.index.array; for (let i = 0; i < ia.length; i++) idx[io + i] = ia[i] + vo;
    vo += g.attributes.position.count; io += ia.length; g.dispose();
  }
  const out = new THREE.BufferGeometry();
  out.setAttribute('position', new THREE.BufferAttribute(pos, 3));
  out.setAttribute('normal', new THREE.BufferAttribute(nor, 3));
  out.setAttribute('uv', new THREE.BufferAttribute(uv, 2));
  out.setIndex(new THREE.BufferAttribute(idx, 1));
  return out;
}

function buildChunk(cx, cz) {
  const rng = mulberry32(hash(cx, cz) + 777);
  const rr = (a, b) => a + rng() * (b - a), ri = (a, b) => Math.floor(rr(a, b + 1)), pick = a => a[Math.floor(rng() * a.length)];
  const ox = cx * C, oz = cz * C;
  const group = new THREE.Group(); group.position.set(ox, 0, oz);
  const buckets = new Map(), colliders = [], updaters = [];
  let base = new THREE.Matrix4();
  const tm = new THREE.Matrix4(), tq = new THREE.Quaternion(), te = new THREE.Euler(), tp = new THREE.Vector3(), ts = new THREE.Vector3();
  const Y = new THREE.Vector3(0, 1, 0);

  function emit(geo, mat, x, y, z, ry = 0, rx = 0, rz = 0, sx = 1, sy = 1, sz = 1) {
    te.set(rx, ry, rz, 'YXZ'); tq.setFromEuler(te);
    tm.compose(tp.set(x, y, z), tq, ts.set(sx, sy, sz)); tm.premultiply(base);
    geo.applyMatrix4(tm);
    let a = buckets.get(mat); if (!a) { a = []; buckets.set(mat, a); } a.push(geo);
  }
  function local(x, y, z, ry, fn) {
    const prev = base;
    const l = new THREE.Matrix4().compose(new THREE.Vector3(x, y, z), new THREE.Quaternion().setFromAxisAngle(Y, ry), new THREE.Vector3(1, 1, 1));
    base = prev.clone().multiply(l); fn(); base = prev;
  }
  function box(w, h, d, sideM, topM, x, y, z, ry = 0, tw = 6.4, th = 6) {
    const [s, t] = boxParts(w, h, d, tw, th); emit(s, sideM, x, y, z, ry); emit(t, topM, x, y, z, ry);
  }
  function cyl(rt, rb, h, seg, mat, x, y, z, rx = 0, ry = 0, rz = 0) { emit(new THREE.CylinderGeometry(rt, rb, h, seg), mat, x, y, z, ry, rx, rz); }
  function col(x, z, w, d, top) { colliders.push({ x0: ox + x - w / 2, x1: ox + x + w / 2, z0: oz + z - d / 2, z1: oz + z + d / 2, top }); }
  function free(x, z, r) { for (const c of colliders) if (x + r > c.x0 - ox && x - r < c.x1 - ox && z + r > c.z0 - oz && z - r < c.z1 - oz) return false; return true; }

  // --- objects ---
  function panel(x, z, bays, floors, rot) {
    const w = bays * 3.2, h = floors * 3, d = 12, fm = pick(facadeMats);
    local(x, 0, z, rot ? Math.PI / 2 : 0, () => {
      box(w, h, d, fm, roofMat, 0, h / 2, 0);
      box(4, 2.6, 3, concreteMat, roofMat, rr(-w / 3, w / 3), h + 1.3, rr(-2, 2), 0, 4, 2.6);
      const n = Math.max(1, Math.floor(w / 12.8));
      for (let i = 0; i < n; i++) {
        const ex = -w / 2 + (i + .5) * (w / n);
        for (const s of [-1, 1]) { box(1.8, 2.4, .2, doorMat, doorMat, ex, 1.2, s * (d / 2 + .1)); box(3.2, .25, 1.4, concreteMat, concreteMat, ex, 2.7, s * (d / 2 + .7)); }
      }
      if (rng() < .55) { const L = Math.min(w * .7, 14); box(L, 2.4, .3, pick(signMats), metalMat, 0, h + 1.2, d * .3, 0, L, 2.4); }
    });
    col(x, z, rot ? d : w, rot ? w : d, h);
  }
  function garages(x, z, n, rot) {

  function house(x, z, rot) {
    const w = 6.4, h = 6, d = 8;
    local(x, 0, z, rot ? Math.PI / 2 : 0, () => {
      box(w, h, d, houseMat, roofMat, 0, h / 2, 0);

  }
  function tree(x, z) {
    const h = rr(4, 8); cyl(.14, .3, h, 6, barkMat, x, h / 2, z);
    const nb = ri(3, 6);
    for (let i = 0; i < nb; i++) {
      const y = rr(h * .4, h * .95), len = rr(1.2, 2.8), g = new THREE.CylinderGeometry(.03, .09, len, 5); g.translate(0, len / 2, 0);
      emit(g, barkMat, x, y, z, rng() * 6.28, 0, rr(.6, 1.2) * (rng() < .5 ? -1 : 1));

// ---------- chunk manager ----------
const chunks = new Map(); let colliders = []; let shadowDirty = true;
function updateChunks(px, pz, force) {
  const pcx = Math.round(px / C), pcz = Math.round(pz / C), RR = CONFIG.VIEW_CELLS; let built = 0, dirty = false;
  for (let r = 0; r <= RR; r++) for (let dx = -r; dx <= r; dx++) for (let dz = -r; dz <= r; dz++) {
    if (Math.max(Math.abs(dx), Math.abs(dz)) !== r) continue;
    const k = (pcx + dx) + ',' + (pcz + dz);
    if (!chunks.has(k)) { if (built >= (force ? 999 : 1)) continue; const ch = buildChunk(pcx + dx, pcz + dz); scene.add(ch.group); chunks.set(k, ch); built++; dirty = true; }
  }
  for (const [k, ch] of chunks) if (Math.max(Math.abs(ch.cx - pcx), Math.abs(ch.cz - pcz)) > RR + 1) {
    scene.remove(ch.group); ch.group.traverse(o => { if (o.geometry) o.geometry.dispose(); }); chunks.delete(k); dirty = true;
  }
  if (dirty) { colliders = [].concat(...[...chunks.values()].map(c => c.colliders)); shadowDirty = true; }
}

let actx = null, noiseBuf = null, scareOn = true;
function scareLoop() {
  if (!scareOn || !actx) return;
  const t = actx.currentTime, k = Math.random();
  const pan = actx.createStereoPanner(); pan.pan.value = Math.random() * 2 - 1; pan.connect(actx.destination);
  if (k < .4) { // distant metal bang
    const s = actx.createBufferSource(); s.buffer = noiseBuf;
    const f = actx.createBiquadFilter(); f.type = 'lowpass'; f.frequency.value = 500;
    const g = actx.createGain(); g.gain.setValueAtTime(.7, t); g.gain.exponentialRampToValueAtTime(.001, t + 1.4);
    const o = actx.createOscillator(); o.type = 'triangle'; o.frequency.setValueAtTime(95 + Math.random() * 60, t); o.frequency.exponentialRampToValueAtTime(40, t + 1.2);
    const og = actx.createGain(); og.gain.setValueAtTime(.35, t); og.gain.exponentialRampToValueAtTime(.001, t + 1.2);
    s.connect(f); f.connect(g); g.connect(pan); o.connect(og); og.connect(pan); s.start(t, Math.random(), 1.5); o.start(t); o.stop(t + 1.3);
  } else if (k < .75) { // creaking metal
    const o = actx.createOscillator(); o.type = 'sawtooth'; const f0 = 180 + Math.random() * 120;
    o.frequency.setValueAtTime(f0, t); o.frequency.linearRampToValueAtTime(f0 * (.6 + Math.random() * .3), t + 1.8);
    const bp = actx.createBiquadFilter(); bp.type = 'bandpass'; bp.frequency.value = 700; bp.Q.value = 8;
    const g = actx.createGain(); g.gain.setValueAtTime(0, t); g.gain.linearRampToValueAtTime(.06, t + .4); g.gain.linearRampToValueAtTime(0, t + 1.8);
    o.connect(bp); bp.connect(g); g.connect(pan); o.start(t); o.stop(t + 1.9);
  } else { // eerie drone swell
    const lp = actx.createBiquadFilter(); lp.type = 'lowpass'; lp.frequency.value = 600;
    const g = actx.createGain(); g.gain.setValueAtTime(0, t); g.gain.linearRampToValueAtTime(.05, t + 2); g.gain.linearRampToValueAtTime(0, t + 5);
    for (const fr of [110, 116.5, 164.8]) { const o = actx.createOscillator(); o.type = 'sine'; o.frequency.value = fr; o.connect(lp); o.start(t); o.stop(t + 5.1); }
    lp.connect(g); g.connect(pan);
  }
  setTimeout(scareLoop, 5000 + Math.random() * 12000);
}
function startAudio() {
  if (actx) { actx.resume(); return; }
  try {
    actx = new (window.AudioContext || window.webkitAudioContext)();
    noiseBuf = actx.createBuffer(1, actx.sampleRate * 2, actx.sampleRate);
    const d = noiseBuf.getChannelData(0); let last = 0;
    for (let i = 0; i < d.length; i++) { last = (last + .02 * (Math.random() * 2 - 1)) / 1.02; d[i] = last * 3.5; }
    const src = actx.createBufferSource(); src.buffer = noiseBuf; src.loop = true;
    const lp = actx.createBiquadFilter(); lp.type = 'lowpass'; lp.frequency.value = 420; lp.Q.value = 1.2;
    const g = actx.createGain(); g.gain.value = .3;
    const lfo = actx.createOscillator(); lfo.frequency.value = .08; const lg = actx.createGain(); lg.gain.value = .15;
    lfo.connect(lg); lg.connect(g.gain); lfo.start();
    src.connect(lp); lp.connect(g); g.connect(actx.destination); src.start();
    setTimeout(scareLoop, 4000);
  } catch (e) { actx = null; }
}
function crunch() {
  if (!actx || !noiseBuf) return;

  const g = actx.createGain(); g.gain.setValueAtTime(.35, actx.currentTime); g.gain.exponentialRampToValueAtTime(.001, actx.currentTime + .11);
  s.connect(f); f.connect(g); g.connect(actx.destination); s.start(0, Math.random() * 1.5, .12);
}

function hitAxis(axis) {
  for (const c of colliders) {
    if (P.y >= c.top - .45) continue;
    if (P.x + RAD > c.x0 && P.x - RAD < c.x1 && P.z + RAD > c.z0 && P.z - RAD < c.z1) {
      if (axis === 'x') { if (P.vx > 0) P.x = c.x0 - RAD; else if (P.vx < 0) P.x = c.x1 + RAD; else P.x = P.x < (c.x0 + c.x1) / 2 ? c.x0 - RAD : c.x1 + RAD; P.vx = 0; }
      else { if (P.vz > 0) P.z = c.z0 - RAD; else if (P.vz < 0) P.z = c.z1 + RAD; else P.z = P.z < (c.z0 + c.z1) / 2 ? c.z0 - RAD : c.z1 + RAD; P.vz = 0; }
    }
  }
}
function stepPlayer(dt) {
  let ix = (keys.KeyD ? 1 : 0) - (keys.KeyA ? 1 : 0), iz = (keys.KeyW ? 1 : 0) - (keys.KeyS ? 1 : 0);
  const len = Math.hypot(ix, iz) || 1; ix /= len; iz /= len;
  const sp = (keys.ShiftLeft || keys.ShiftRight) ? CONFIG.SPRINT : CONFIG.WALK, s = Math.sin(yaw), c = Math.cos(yaw);
  const tx = (ix * c - iz * s) * sp, tz = (-ix * s - iz * c) * sp, k = 1 - Math.exp(-(P.ground ? 12 : 2.5) * dt);
  P.vx += (tx - P.vx) * k; P.vz += (tz - P.vz) * k;
  P.x += P.vx * dt; hitAxis('x'); P.z += P.vz * dt; hitAxis('z');
  if (keys.Space && P.ground) { P.vy = CONFIG.JUMP; P.ground = false; }
  P.vy -= CONFIG.GRAVITY * dt; P.y += P.vy * dt;
  let gy = 0;
  for (const cc of colliders) if (cc.top <= P.y + .45 && cc.top > gy && P.x + RAD > cc.x0 && P.x - RAD < cc.x1 && P.z + RAD > cc.z0 && P.z - RAD < cc.z1) gy = cc.top;
  if (P.y <= gy) { P.y = gy; if (P.vy < 0) P.vy = 0; P.ground = true; } else P.ground = false;
  if (P.y < -50) { P.x = -6; P.y = 0; P.z = 22; P.vy = 0; }
  const spd = Math.hypot(P.vx, P.vz);
  if (P.ground && spd > .5) { bobT += spd * dt * 1.9; stepAcc += spd * dt; if (stepAcc > 1.7) { stepAcc = 0; crunch(); } }
  return spd;
}

// ---------- main loop ----------
let shadowT = 0, lsx = 1e9, lsz = 1e9;
let last = performance.now(), fps = 60, fAcc = 0, fN = 0, raf = 0;
const moon = new THREE.Color(0x5a6a99), warm = new THREE.Color(0xffd2a0), tmpC = new THREE.Color(), NIGHT_FOG = new THREE.Color('#06090e');
let flickT = 0, flickV = 1;
function frame(now) {
  raf = requestAnimationFrame(frame);
  const dt = Math.min(.05, (now - last) / 1000); last = now;
  fAcc += dt; fN++; if (fAcc > .5) { fps = Math.round(fN / fAcc); fAcc = 0; fN = 0; }
  const spd = locked ? stepPlayer(dt) : (P.vx = P.vz = 0);
  updateChunks(P.x, P.z, false);
  for (const ch of chunks.values()) for (const u of ch.updaters) u(dt);

  // day / night
  day += (dayT - day) * Math.min(1, dt * 1.5);
  scene.fog.color.lerpColors(NIGHT_FOG, HORIZON, day); scene.background.copy(scene.fog.color);
  skyMat.color.setRGB(scene.fog.color.r / HORIZON.r, scene.fog.color.g / HORIZON.g, scene.fog.color.b / HORIZON.b);
  scene.fog.density = CONFIG.FOG * (1.5 - .5 * day);
  sun.intensity = CONFIG.SUN * (.05 + .95 * day); hemi.intensity = CONFIG.HEMI * (.17 + .83 * day);
  sun.color.copy(tmpC.lerpColors(moon, warm, day));
  // creepy flickering flashlight
  if (flickT > 0) { flickT -= dt; flickV = Math.random() < .35 ? 0 : Math.random(); }
  else { flickV = 1; if (Math.random() < .004) flickT = .1 + Math.random() * .5; }
  flash.intensity = flashOn ? 320 * flickV : 0;

  // camera
  const bob = P.ground ? Math.sin(bobT) * .04 * Math.min(1, spd / 5) : 0;
  camera.position.set(P.x, P.y + 1.62 + bob, P.z); camera.rotation.set(pitch, yaw, 0);
  sky.position.copy(camera.position);
  const sx = Math.round(P.x / 4) * 4, sz = Math.round(P.z / 4) * 4;
  sun.target.position.set(sx, 0, sz); sun.position.set(sx + SUN_OFF.x, SUN_OFF.y, sz + SUN_OFF.z);
  shadowT += dt;
  if (shadowDirty || shadowT > .3 || sx !== lsx || sz !== lsz) { renderer.shadowMap.needsUpdate = true; shadowDirty = false; shadowT = 0; lsx = sx; lsz = sz; }

  // snow
  const p = snowGeo.attributes.position.array, t = now * .001;
  for (let i = 0; i < NS; i++) {
    const j = i * 3; p[j + 1] -= snowFall[i] * dt; p[j] += (1.2 + Math.sin(t * .7 + i) * .4) * dt;
    if (p[j + 1] < 0) p[j + 1] += 25;
    if (p[j] - P.x > 40) p[j] -= 80; else if (p[j] - P.x < -40) p[j] += 80;
    if (p[j + 2] - P.z > 40) p[j + 2] -= 80; else if (p[j + 2] - P.z < -40) p[j + 2] += 80;
  }
  snowGeo.attributes.position.needsUpdate = true;

  hud.textContent = `${fps} fps   x ${P.x.toFixed(0)}  z ${P.z.toFixed(0)}\n${day > .5 ? 'day' : 'night'} (N)   flashlight ${flashOn ? 'on' : 'off'} (F)`;
  renderer.render(scene, camera);
}
raf = requestAnimationFrame(frame);

window.__RU_CITY_STOP = () => {
  cancelAnimationFrame(raf); ac.abort(); scareOn = false;
  if (document.pointerLockElement) document.exitPointerLock();
  try { actx && actx.close(); } catch (e) {}
  renderer.dispose(); wrap.remove();
};
console.log('%cABANDONED CITY loaded — press PLAY. Stop it with __RU_CITY_STOP()', 'color:#e8d9b8;font-size:13px');
})();
