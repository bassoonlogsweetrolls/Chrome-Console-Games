function evalPass(p, m, ax, ay, cone) {
  const d = G.teams[p.team].dir;
  const dx = m.x - p.x, dy = m.y - p.y, dist = hyp(dx, dy);
  if (dist < 3.5 || dist > 46) return -1e9;
  let cos = 1;
  if (ax !== undefined) { cos = (dx * ax + dy * ay) / dist; if (cos < cone) return -1e9; }
  let open = 14, blocked = 0;
  for (const o of G.teams[1 - p.team].players) {
    if (o.out) continue;
    const od = hyp(o.x - m.x, o.y - m.y); if (od < open) open = od;
    if (segDist(o, p, m) < 1.7 && hyp(o.x - p.x, o.y - p.y) > 1) blocked++;
  }
  let s = (m.x - p.x) * d * 0.3 + open * 1.2 - blocked * 11 - dist * 0.24 + (dist < 22 ? 4 : 0) + (ax !== undefined ? cos * 22 : 0);
  if (m.role === 'GK') s -= 9;
  return s;
}
function bestTarget(p, ax, ay, cone, extra) {
  let best = null, bs = -1e9;
  for (const m of G.teams[p.team].players) {
    if (m === p || m.out) continue;
    let s = evalPass(p, m, ax, ay, cone);
    if (extra) s += extra(m);
    if (s > bs) { bs = s; best = m; }
  }
  return bs > -1e8 ? { m: best, s: bs } : null;
}

  const d0 = hyp(tx - p.x, ty - p.y);
  if (kind === 'through') { tx += d * 8 + m.vx * 0.8; ty += m.vy * 0.8; } else { const t = d0 / 16; tx += m.vx * t * 0.8; ty += m.vy * t * 0.8; }
  tx = clamp(tx, 1, PW - 1); ty = clamp(ty, 1, PH - 1);
  const dx = tx - p.x, dy = ty - p.y, dist = Math.max(0.1, hyp(dx, dy));
  const ang = Math.atan2(dy, dx) + gauss() * (0.012 + (1 - p.r / 100) * 0.075) * (1 + dist / 45) * (power > 0.9 ? 1.5 : 1);
  const ux = Math.cos(ang), uy = Math.sin(ang);
  if (kind === 'lob') {
    const vh = clamp(dist * 0.85 + 7, 11, 27) * (0.92 + power * 0.2);
    kickBall(p, ux, uy, vh, GRAV * (dist / vh) / 2, { pass: 1, target: m });
  } else {
    const v = clamp(Math.sqrt(81 + 6.4 * dist), 10, 28) * (0.88 + power * 0.3);
    kickBall(p, ux, uy, v, 0, { pass: 1, target: m });
  }
  G.teams[p.team].stats.passes++;
}
function lobTo(p, tx, ty, target) {
  const dx = tx - p.x, dy = ty - p.y, dist = Math.max(1, hyp(dx, dy));
  const ang = Math.atan2(dy, dx) + gauss() * 0.05;
  const vh = clamp(dist * 0.85 + 7, 11, 28);
  kickBall(p, Math.cos(ang), Math.sin(ang), vh, GRAV * (dist / vh) / 2, { pass: target ? 1 : 0, target });
  if (target) G.teams[p.team].stats.passes++;
}
function doShoot(p, aimY, power, pen) {
  const T = G.teams[p.team], gx = T.dir > 0 ? PW : 0;
  const ty = CY + clamp(aimY, -1, 1) * (GW / 2 - 0.3);
  const dx = gx - p.x, dy = ty - p.y, dist = hyp(dx, dy);
  const press = (() => { let m = 99; for (const o of G.teams[1 - p.team].players) if (!o.out) m = Math.min(m, hyp(o.x - p.x, o.y - p.y)); return m < 2.5; })();
  const sig = 0.03 + dist * 0.0024 * (1.35 - p.r / 100) + (power > 0.92 ? 0.05 : 0) + (press ? 0.035 : 0) + (pen ? -0.02 : 0);
  const ang = Math.atan2(dy, dx) + gauss() * sig * (pen ? 0.7 : 1);
  const sp = 21 + power * 13;
  const vz = rnd(0, 1.6) + (power > 0.85 ? rnd(0, 5) : 0) + (dist > 25 ? rnd(0, 1.5) : 0);
  kickBall(p, Math.cos(ang), Math.sin(ang), sp, vz, { shot: 1 });
  T.stats.shots++;
  const b = G.ball, t = Math.abs((gx - b.x) / (b.vx || 1e-6)), yl = b.y + b.vy * t, zl = vz * t - 0.5 * GRAV * t * t;
  if (Math.abs(yl - CY) < GW / 2 && (zl < GH)) T.stats.sot++;
}

// ───────────────────────── Entradas, faltas y tarjetas ─────────────────────────
function startTackle(p, slide) {
  if (p.tackleT > 0 || p.slideT > 0 || p.tkCool > 0 || p.stun > 0 || p.diveT > 0) return;
  p.tkRes = false; p.tkCool = 1.1;
  if (slide) {
    p.slideT = 0.7; const sp = Math.max(hyp(p.vx, p.vy), 6) + 3.5;
    p.vx = Math.cos(p.face) * sp; p.vy = Math.sin(p.face) * sp;
    emit(8, p.x, p.y, { s0: 1, s1: 4, col: ['#6b8e3a', '#4a6d2a', '#8a6a3a'], l0: .3, l1: .7, vz0: 1, vz1: 3, g: 12, kind: 'dot' });
  } else {
    p.tackleT = 0.33; p.vx += Math.cos(p.face) * 3; p.vy += Math.sin(p.face) * 3;
  }
}
function resolveTackles() {
  const b = G.ball, o = b.owner;
  for (const p of G.all) {
    if (p.out || p.tkRes || !(p.tackleT > 0 || p.slideT > 0)) continue;
    if (!o || o.team === p.team) { if (p.tackleT <= 0 && p.slideT <= 0) p.tkRes = true; continue; }
    const slide = p.slideT > 0;
    if (hyp(b.x - p.x, b.y - p.y) > (slide ? 1.9 : 1.5) && hyp(o.x - p.x, o.y - p.y) > (slide ? 1.9 : 1.5)) continue;
    p.tkRes = true;
    const angTo = Math.atan2(p.y - o.y, p.x - o.x), front = Math.abs(angDiff(o.face, angTo)) < 1.2, behind = Math.abs(angDiff(o.face, angTo)) > 1.9;
    let ps = 0.42 + (p.r - o.r) / 160 + (slide ? 0.12 : 0) + (front ? 0.08 : -0.1);
    if (!G.teams[p.team].ctrl) ps += (G.diff - 1) * 0.06;
    if (G.teams[o.team].ctrl && !G.teams[p.team].ctrl) ps -= 0.04;
    const pf = (behind ? 0.5 : front ? 0.1 : 0.25) * (slide ? 1.5 : 1) * (!G.teams[p.team].ctrl ? 0.8 : 1);
    const r = Math.random();
    if (r < ps) {
      G.teams[p.team].stats.tackles++; p.tackles++;
      if (Math.random() < pf * 0.3) { callFoul(p, o, slide, behind); return; }
      o.stun = 0.5; o.cool = 0.5;
      if (Math.random() < 0.5) setOwner(p);
      else { b.owner = null; const a = Math.atan2(p.vy, p.vx) + rnd(-0.8, 0.8); b.vx = Math.cos(a) * rnd(4, 9); b.vy = Math.sin(a) * rnd(4, 9); b.vz = rnd(0, 2); b.last = p; b.lastT = p.team; b.passTeam = -1; G.possTeam = p.team; }
      Snd.kick(0.4);
      return;
    } else if (Math.random() < pf) { callFoul(p, o, slide, behind); return; }
  }
}
function callFoul(p, o, slide, behind) {
  const T = G.teams[p.team]; T.stats.fouls++;
  Snd.whistle(false);
  let card = 0; const r = Math.random();
  const redP = slide && behind ? 0.07 : 0.012, yelP = (slide ? 0.4 : 0.12) * (behind ? 1.6 : 1);
  if (r < redP) card = 2; else if (r < redP + yelP) card = 1;
  if (card === 1) { p.yellow++; T.stats.yellow++; if (p.yellow >= 2) card = 3; }
  if (card >= 2) { T.stats.red++; if (card === 3) T.stats.yellow--; }
  G.card = card ? { x: p.x, y: p.y, kind: card === 1 ? 'y' : card === 2 ? 'r' : 'yr', t: 0, name: p.name } : null;
  const spotX = o.x, spotY = o.y;
  const inBox = (T.dir > 0 ? spotX < BOX_L : spotX > PW - BOX_L) && Math.abs(spotY - CY) < BOX_W / 2;
  if (card) {
    banner(card === 1 ? 'TARJETA AMARILLA' : 'TARJETA ROJA', `${p.name} (${T.def.s})`, card === 1 ? '#facc15' : '#ef4444', 2.2);
    ticker(`${gameMin()}' ${card === 1 ? '🟨' : '🟥'} ${p.name} (${T.def.s})`, card === 1 ? '#facc15' : '#ef4444');
    if (card >= 2) { p.out = true; p.x = -30; p.y = -30; p.vx = p.vy = 0; const c = G.hum.find(h => h.player === p); if (c) c.player = null; }
  } else banner(inBox ? '¡PENALTI!' : 'FALTA', `${p.name} (${T.def.s})`, '#ffffff', 1.6);
  if (inBox) setRestart('pen', o.team, G.teams[o.team].dir > 0 ? PW - PEN : PEN, CY);
  else setRestart('free', o.team, clamp(spotX, 2, PW - 2), clamp(spotY, 1.5, PH - 1.5));
}

function setKickoff(team) {
  const b = G.ball; b.x = PW / 2; b.y = CY; b.owner = null; b.vx = b.vy = b.vz = 0; b.z = 0;
  G.possTeam = team;
  setRestart('kickoff', team, PW / 2, CY);
}
function setRestart(type, team, x, y) {
  const b = G.ball, T = G.teams[team], d = T.dir, goalX = d > 0 ? PW : 0;
  G.phase = 'restart';
  const R = G.restart = { type, team, x, y, t: 0, wait: 1.2, taker: null, goalX };
  b.owner = null; b.vx = b.vy = b.vz = 0; b.z = 0; b.shot = -1; b.target = null; b.passTeam = -1; b.x = x; b.y = y;
  let cands = T.players.filter(p => !p.out), taker = null;
  const nearest = (list, px, py) => list.reduce((a, c) => (!a || hyp(c.x - px, c.y - py) < hyp(a.x - px, a.y - py)) ? c : a, null);
  if (type === 'goalkick') taker = cands.find(p => p.role === 'GK');
  else if (type === 'pen') { taker = cands.filter(p => p.role !== 'GK').sort((a, c) => c.r - a.r)[0]; R.wait = 1.8; }
  else if (type === 'free' && hyp(goalX - x, CY - y) < 32) { taker = cands.filter(p => p.role !== 'GK').sort((a, c) => c.r - a.r)[0]; R.wait = 1.6; }
  else if (type === 'kickoff') { taker = cands.filter(p => p.role === 'FWD').sort((a, c) => G.teams[team].form.pos[c.k][0] - G.teams[team].form.pos[a.k][0])[0]; }
  else taker = nearest(cands.filter(p => p.role !== 'GK'), x, y);
  if (type === 'corner') R.wait = 1.6;
  if (!taker) taker = cands[0];
  R.taker = taker;
  // posiciones objetivo del resto
  for (const p of G.all) {
    if (p.out) continue;
    p.vx = p.vy = p.wx = p.wy = 0; p.slideT = p.tackleT = p.diveT = 0; p.stun = 0; p.cool = 0;
    if (p === taker) continue;
    const t2 = G.teams[p.team], dd = t2.dir;
    let [tx, ty] = formPos(p);
    if (type === 'kickoff') { tx = dd > 0 ? Math.min(tx, PW / 2 - 2) : Math.max(tx, PW / 2 + 2); }
    if (type === 'pen') {
      if (p.role === 'GK') { if (p.team !== team) { tx = goalX > PW / 2 ? PW - 0.8 : 0.8; ty = CY; } }
      else { tx = goalX > PW / 2 ? PW - BOX_L - 3 - (p.k % 5) * 1.6 : BOX_L + 3 + (p.k % 5) * 1.6; ty = 6 + (p.k * 5.3) % 56; }
    } else if (p.team === team && type === 'corner' && p.role !== 'GK' && p.k >= 4) {
      tx = goalX + (goalX > PW / 2 ? -1 : 1) * rnd(5, 13); ty = CY + rnd(-9, 9);
    } else if (p.team !== team && p.role !== 'GK' && type === 'corner') {
      tx = goalX + (goalX > PW / 2 ? -1 : 1) * rnd(3, 11); ty = CY + rnd(-8, 8);
    } else if (type === 'goalkick' && p.team !== team && p.role !== 'GK') {
      const lim = goalX > PW / 2 ? PW - BOX_L - 2 : BOX_L + 2; tx = goalX > PW / 2 ? Math.min(tx, lim) : Math.max(tx, lim);
    }
    const kr = type === 'throw' ? 3.5 : (type === 'free' || type === 'corner') ? CIRC : type === 'kickoff' ? CIRC : 0;
    if (kr && p.team !== team) {
      const dx = tx - x, dy = ty - y, dd2 = hyp(dx, dy) || 0.01;
      if (dd2 < kr) { tx = x + dx / dd2 * kr; ty = y + dy / dd2 * kr; }
    }
    p.rtx = clamp(tx, 1, PW - 1); p.rty = clamp(ty, 1, PH - 1);
    if (type === 'kickoff' || type === 'pen') { p.x = p.rtx; p.y = p.rty; p.face = dd > 0 ? 0 : Math.PI; }
    else if (hyp(p.x - p.rtx, p.y - p.rty) > 45) { p.x = p.rtx; p.y = p.rty; }
  }
  // el que saca
  const tx0 = type === 'throw' ? x : x, ty0 = type === 'throw' ? (y < CY ? -0.3 : PH + 0.3) : y;
  taker.x = tx0 + (type === 'corner' ? (x < PW / 2 ? -0.4 : 0.4) : 0); taker.y = ty0;
  if (type === 'free' || type === 'pen') { taker.x = x - d * 1.0; taker.y = y; }
  taker.vx = taker.vy = 0;
  taker.face = type === 'throw' ? (y < CY ? Math.PI / 2 : -Math.PI / 2) : type === 'corner' ? Math.atan2(CY - y, goalX - x) : type === 'goalkick' ? (d > 0 ? 0 : Math.PI) : (d > 0 ? 0 : Math.PI);
  setOwner(taker);
  b.x = x; b.y = type === 'throw' ? (y < CY ? 0 : PH) : y;
  if (type === 'throw') { b.x = x; }
  const c = G.hum.find(h => h.team === team);
  if (c && taker.role !== 'GK') c.player = taker;
  const names = { throw: 'SAQUE DE BANDA', corner: 'SAQUE DE ESQUINA', goalkick: 'SAQUE DE PUERTA', free: 'FALTA', pen: 'PENALTI', kickoff: '' };
  if (type === 'corner') T.stats.corners++;
  if (names[type] && type !== 'free' && type !== 'pen') { banner(names[type], T.def.n, T.kit.main, 1.4); }
  if (type === 'kickoff' && G.clock > 0 && !G.demo) Snd.whistle(false);
  if (type === 'pen' || type === 'free') Snd.whistle(false);
}
function restartAI(p) {
  const R = G.restart; if (!R || p === R.taker || p.stun > 0) return;
  goto(p, p.rtx, p.rty, 0.85, true);
}
function aiRestartKick() {
  const R = G.restart, p = R.taker, T = G.teams[p.team], d = T.dir, gk = G.teams[1 - p.team].players.find(q => q.role === 'GK');
  const aimSide = () => { const gy = gk ? gk.y : CY; return (gy > CY ? -1 : 1) * rnd(0.45, 0.9); };
  switch (R.type) {
    case 'pen': doShoot(p, aimSide(), rnd(0.55, 0.95), true); break;
    case 'free': {
      const dg = hyp(R.goalX - p.x, CY - p.y);
      if (dg < 32 && p.role !== 'GK') { doShoot(p, aimSide(), rnd(0.6, 0.95)); break; }
      const t = bestTarget(p, d, 0, -0.2); if (t) doPass(p, t.m, hyp(t.m.x - p.x, t.m.y - p.y) > 24 ? 'lob' : 'ground', 0.4); else lobTo(p, p.x + d * 25, CY);
      break;
    }
    case 'corner': {
      const tgt = G.teams[p.team].players.filter(q => !q.out && q.role !== 'GK' && q !== p && Math.abs(q.x - R.goalX) < 20).sort((a, c) => Math.abs(a.y - CY) - Math.abs(c.y - CY))[0];
      if (tgt) lobTo(p, tgt.x, tgt.y, tgt); else lobTo(p, R.goalX + (R.goalX > PW / 2 ? -9 : 9), CY + rnd(-5, 5));
      break;
    }
    case 'goalkick': {
      const t = bestTarget(p, d, 0, 0.2, m => m.role === 'MID' ? 8 : 0);
      if (t) lobTo(p, t.m.x, t.m.y, t.m); else lobTo(p, p.x + d * 35, CY + rnd(-12, 12));
      break;
    }
    case 'throw': {
      const t = bestTarget(p, 0, (p.y < CY ? 1 : -1), -0.3); if (t) doPass(p, t.m, 'ground', 0.1); else lobTo(p, p.x + d * 10, CY);
      break;
    }
    default: { const t = bestTarget(p, -d, 0, -1); if (t) doPass(p, t.m, 'ground', 0.1); else lobTo(p, p.x - d * 10, CY); }
  }
}

// ───────────────────────── IA ─────────────────────────
function goto(p, x, y, frac, spr) {
  const dx = x - p.x, dy = y - p.y, d = hyp(dx, dy);
  if (d < 0.35) { p.wx = p.wy = 0; return d; }
  const k = Math.min(1, d / 2.5) * frac;
  p.wx = dx / d * k; p.wy = dy / d * k; p.sprint = !!spr && d > 6;
  return d;
}
function computeRanks() {
  const b = G.ball;
  for (const T of G.teams) {
    let rx, ry;
    if (b.owner && b.owner.team !== T.i) { rx = b.owner.x; ry = b.owner.y; } else { rx = b.x + b.vx * 0.4; ry = b.y + b.vy * 0.4; }
    for (const p of T.players) p.rank = 99;
    outfield(T).filter(p => !p.ctl).map(p => ({ p, d: hyp(p.x - rx, p.y - ry) - (b.target === p ? 4 : 0) }))
      .sort((a, c) => a.d - c.d).forEach((e, i) => { e.p.rank = i; });
  }
}
function aiControl(p) {
  p.wx = p.wy = 0; p.sprint = false;
  const ph = G.phase;
  if (p.out) return;
  if (ph === 'goal') return celebrate(p);
  if (ph === 'restart') return restartAI(p);
  if (ph !== 'play') return;
  if (p.stun > 0 || p.slideT > 0 || p.diveT > 0 || p.tackleT > 0) return;
  const dt = DT;
  if (p.role === 'GK') return gkAI(p, dt);
  const own = G.ball.owner;
  if (own === p) return carrierAI(p, dt);
  if (!own) return looseAI(p, dt);
  if (own.team === p.team) return supportAI(p, dt);
  return defendAI(p, dt);
}
function celebrate(p) {
  const gi = G.goalInfo; if (!gi || !gi.player) return;
  if (p === gi.player) {
    const cx = gi.left ? 6 : PW - 6, cy = gi.player.y < CY ? 2 : PH - 2;
    goto(p, cx, cy, 1, true);
  } else if (p.team === gi.player.team && p.role !== 'GK' && G.t - gi.t0 > 0.8) goto(p, gi.player.x, gi.player.y, 0.9, true);
}
function looseAI(p) {
  const b = G.ball;
  if (p.rank < 2) goto(p, b.x + b.vx * 0.35, b.y + b.vy * 0.35, 1, true);
  else { const f = formPos(p); goto(p, f[0], f[1], 0.9, true); }
}
function supportAI(p) {
  let [tx, ty] = formPos(p);
  const d = G.teams[p.team].dir;
  if (p.role === 'FWD') tx += d * 5; else if (p.role === 'MID') tx += d * 2;
  for (const m of G.teams[p.team].players) {
    if (m === p || m.out) continue;
    const dd = hyp(m.x - p.x, m.y - p.y);
    if (dd < 6 && dd > 0.01) { tx += (p.x - m.x) / dd * (6 - dd) * 0.6; ty += (p.y - m.y) / dd * (6 - dd) * 0.6; }
  }
  goto(p, clamp(tx, 1, PW - 1), clamp(ty, 2, PH - 2), 0.9, true);
}
function defendAI(p, dt) {
  const T = G.teams[p.team], d = T.dir, own = G.ball.owner;
  const og = { x: d > 0 ? 0 : PW, y: CY };
  p.tkT -= dt;
  if (p.rank === 0) {
    const dd = goto(p, own.x + own.vx * 0.3, own.y + own.vy * 0.3, 1, true);
    if (dd < 1.7 && p.tkCool <= 0 && p.tkT <= 0) {
      p.tkT = rnd(0.35, 0.8);
      const base = [0.45, 0.6, 0.75][G.diff];
      const pr = (G.teams[1 - p.team].ctrl ? base : 0.6) * (p.r / 80);
      if (Math.random() < pr) { p.face = Math.atan2(own.y - p.y, own.x - p.x); startTackle(p, dd > 1.35 && Math.random() < 0.12); }
    }
    return;
  }
  if (p.rank === 1) {
    const dx = og.x - own.x, dy = og.y - own.y, l = hyp(dx, dy) || 1;
    goto(p, own.x + dx / l * 5, own.y + dy / l * 5, 1, true); return;
  }
  const fp = formPos(p); let best = null, bd = 14;
  for (const o of G.teams[1 - p.team].players) {
    if (o.out || o.role === 'GK' || o === own) continue;
    const dd = hyp(o.x - fp[0], o.y - fp[1]);
    if (dd < bd) { bd = dd; best = o; }
  }
  if (best) {
    const dx = og.x - best.x, dy = og.y - best.y, l = hyp(dx, dy) || 1;
    goto(p, lerp(fp[0], best.x + dx / l * 2.6, 0.7), lerp(fp[1], best.y + dy / l * 2.6, 0.7), 0.95, true);
  } else goto(p, fp[0], fp[1], 0.9, true);
}
function aiShoot(p) {
  const gk = G.teams[1 - p.team].players.find(q => q.role === 'GK' && !q.out);
  doShoot(p, (gk && gk.y > CY ? -1 : 1) * rnd(0.45, 0.92), rnd(0.6, 0.95));
}
function carrierAI(p, dt) {
  const T = G.teams[p.team], d = T.dir, gx = d > 0 ? PW : 0, opps = G.teams[1 - p.team].players;
  p.holdT += dt; p.think -= dt;
  let near = 99;
  for (const o of opps) { if (o.out) continue; const dd = hyp(o.x - p.x, o.y - p.y); if (dd < near) near = dd; }
  const dg = hyp(gx - p.x, CY - p.y);
  let dx = gx - p.x, dy = (CY - p.y) * 0.9; const l = hyp(dx, dy) || 1; dx /= l; dy /= l;
  for (const o of opps) {
    if (o.out || o.role === 'GK') continue;
    const ox = p.x - o.x, oy = p.y - o.y, od = hyp(ox, oy);
    if (od < 6.5 && od > 0.01) { const w = (6.5 - od) / 6.5 * 2.1; dx += ox / od * w; dy += oy / od * w; }
  }
  if (p.y < 4) dy += 0.6; if (p.y > PH - 4) dy -= 0.6;
  const l2 = hyp(dx, dy) || 1;
  p.wx = dx / l2; p.wy = dy / l2; p.sprint = near > 2.4 && p.stam > 0.15;
  if (p.think > 0) return;
  p.think = rnd(0.16, 0.3);
  const maxD = p.role === 'FWD' ? 32 : p.role === 'MID' ? 27 : 19;
  if (dg < maxD && p.holdT > 0.2) {
    let clear = true;
    for (const o of opps) if (!o.out && o.role !== 'GK' && segDist(o, p, { x: gx, y: CY }) < 1.5 && hyp(o.x - p.x, o.y - p.y) < dg) { clear = false; break; }
    const pr = (dg < 12 ? 0.7 : dg < 20 ? 0.45 : 0.22) * (clear ? 1 : 0.35) * (p.r / 80);
    if (Math.random() < pr) { aiShoot(p); return; }
  }
  const pressure = near < 3.2, t = bestTarget(p);
  if (t) {
    const dist = hyp(t.m.x - p.x, t.m.y - p.y);
    if ((pressure && t.s > -6) || (t.s > 20 && p.holdT > 0.9) || p.holdT > 3.2) {
      const kind = dist > 30 ? 'lob' : (t.m.role === 'FWD' && Math.random() < 0.25 && (t.m.x - p.x) * d > 8) ? 'through' : 'ground';
      doPass(p, t.m, kind, 0.4); return;
    }
  }
  if (pressure && near < 1.6 && p.holdT > 1.5) lobTo(p, p.x + d * 30, clamp(CY + rnd(-14, 14), 3, PH - 3));
}

function gkDiveTo(gk, ty, time) {
  gk.diveT = 0.9; gk.diveTime = time; gk.diveCool = 1.3;
  const dy = ty - gk.y, k = clamp(Math.abs(dy) / time, 0, 17) * Math.sign(dy || 1);
  gk.vx = 0; gk.vy = k; gk.face = dy >= 0 ? Math.PI / 2 : -Math.PI / 2;
}
function gkAI(p, dt) {
  const T = G.teams[p.team], b = G.ball, d = T.dir, gx = d > 0 ? 0 : PW;
  if (b.owner === p) {
    p.holdT += dt;
    if (p.holdT > 1.3) {
      const t = bestTarget(p, d, 0, -0.3, m => m.role === 'DEF' ? 6 : 0);
      if (t) doPass(p, t.m, hyp(t.m.x - p.x, t.m.y - p.y) > 24 ? 'lob' : 'ground', 0.5); else lobTo(p, p.x + d * 40, CY + rnd(-15, 15));
    }
    return;
  }
  const dBall = hyp(b.x - gx, b.y - CY) || 1, ld = clamp(dBall * 0.1, 1.1, 5);
  let tx = gx + (b.x - gx) / dBall * ld, ty = clamp(CY + (b.y - CY) / dBall * ld * 1.2, CY - GW / 2 - 0.4, CY + GW / 2 + 0.4), spr = false;
  const sp = hyp(b.vx, b.vy);
  const inBox = (d > 0 ? b.x < BOX_L : b.x > PW - BOX_L) && Math.abs(b.y - CY) < BOX_W / 2;
  if (!b.owner && inBox && sp < 9 && hyp(b.x - p.x, b.y - p.y) < 14 && b.lastT !== p.team) { tx = b.x; ty = b.y; spr = true; }
  if (!b.owner && b.lastT !== p.team && sp > 8 && (d > 0 ? b.vx < -2 : b.vx > 2) && p.diveCool <= 0 && G.phase === 'play') {
    const t = (p.x - b.x) / b.vx;
    if (t > 0 && t < 0.62) {
      const yl = b.y + b.vy * t, zl = b.z + b.vz * t - 0.5 * GRAV * t * t;
      if (Math.abs(yl - CY) < GW / 2 + 1 && zl < 2.8 && zl > -0.5) {
        const need = Math.abs(yl - p.y);
        let ok = 0.38 + p.r / 220 - Math.max(0, sp - 16) * 0.014 - (need > 2 ? 0.22 : 0) - (zl > 1.9 ? 0.16 : 0) - (zl > 2.4 ? 0.1 : 0);
        if (!T.ctrl) ok += [-0.1, 0, 0.06][G.diff];
        p.diveCool = 1.3;
        if (need < 0.9) { p.saveOK = true; ty = yl; spr = true; }
        else {
          p.saveOK = Math.random() < clamp(ok, 0.06, 0.93);
          const target = p.saveOK ? yl : yl + (yl > p.y ? 1 : -1) * rnd(2.4, 3.8) * (Math.random() < 0.5 ? 1 : -0.6);
          gkDiveTo(p, target, clamp(t * 0.8, 0.18, 0.5));
          return;
        }
      }
    }
  }
  goto(p, tx, ty, 1, spr);
}

// ───────────────────────── Movimiento y físicas ─────────────────────────
function movePlayer(p, dt) {
  if (p.out) return;
  if (p.slideT > 0) { const f = Math.exp(-2.6 * dt); p.vx *= f; p.vy *= f; }
  else if (p.diveT > 0) { if (0.9 - p.diveT > p.diveTime) { const f = Math.exp(-9 * dt); p.vx *= f; p.vy *= f; } }
  else if (p.tackleT > 0) { const f = Math.exp(-3 * dt); p.vx *= f; p.vy *= f; }
  else {
    let wx = p.wx, wy = p.wy; const mag = hyp(wx, wy);
    if (mag > 1) { wx /= mag; wy /= mag; }
    const canSprint = p.sprint && !p.tired;
    let maxv = p.spd * (canSprint ? 1 : 0.64) * (p.ctl || G.teams[p.team].ctrl ? 1 : DIFFK[G.diff]) * (G.ball.owner === p ? 0.94 : 1);
    if (p.stun > 0) maxv *= 0.25;
    let dvx = wx * maxv - p.vx, dvy = wy * maxv - p.vy; const dl = hyp(dvx, dvy), md = 34 * dt;
    if (dl > md) { dvx *= md / dl; dvy *= md / dl; }
    p.vx += dvx; p.vy += dvy;
    if (canSprint && mag > 0.1 && hyp(p.vx, p.vy) > 4) p.stam -= 0.17 * dt; else p.stam = Math.min(1, p.stam + 0.08 * dt);
    if (p.stam <= 0) { p.tired = true; p.stam = 0; } if (p.tired && p.stam > 0.3) p.tired = false;
  }
  p.x += p.vx * dt; p.y += p.vy * dt;
  const sp = hyp(p.vx, p.vy);
  if (p.slideT <= 0 && p.diveT <= 0 && p.tackleT <= 0 && sp > 0.8) {
    let tg = Math.atan2(p.vy, p.vx);
    if (p.ctl && hyp(p.wx, p.wy) > 0.1) tg = Math.atan2(p.wy, p.wx);
    p.face += clamp(angDiff(p.face, tg), -14 * dt, 14 * dt);
  }
  p.ph += sp * dt * 1.5;
  p.x = clamp(p.x, -3, PW + 3); p.y = clamp(p.y, -3, PH + 3);
}
function separation() {
  const a = G.all;
  for (let i = 0; i < a.length; i++) {
    const p = a[i]; if (p.out || p.slideT > 0 || p.diveT > 0) continue;
    for (let j = i + 1; j < a.length; j++) {
      const q = a[j]; if (q.out || q.slideT > 0 || q.diveT > 0) continue;
      const dx = q.x - p.x, dy = q.y - p.y, d = hyp(dx, dy);
      if (d < 0.85 && d > 0.001) { const f = (0.85 - d) / 2 / d; p.x -= dx * f; p.y -= dy * f; q.x += dx * f; q.y += dy * f; }
    }
  }
}
function ballUpdate(dt) {
  const b = G.ball;
  if (b.owner) {
    const o = b.owner, sp = hyp(o.vx, o.vy), lead = 0.55 + 0.5 * Math.min(1, sp / 8);
    const tx = o.x + Math.cos(o.face) * lead, ty = o.y + Math.sin(o.face) * lead, k = 1 - Math.exp(-dt * 14);
    b.x += (tx - b.x) * k; b.y += (ty - b.y) * k; b.vx = o.vx; b.vy = o.vy; b.z = 0; b.vz = 0; b.rot += sp * dt * 2;
    if (G.phase === 'restart') { b.x = G.restart.x; b.y = G.restart.type === 'throw' ? (G.restart.y < CY ? 0 : PH) : G.restart.y; }
    return;
  }
  const n = 2, h = dt / n;
  for (let i = 0; i < n; i++) {
    const ox = b.x, oy = b.y;
    b.x += b.vx * h; b.y += b.vy * h;
    if (b.z > 0 || b.vz > 0) {
      b.z += b.vz * h; b.vz -= GRAV * h;
      if (b.z <= 0) { b.z = 0; if (b.vz < -3) { Snd.bounce(clamp(-b.vz / 12, 0.2, 1)); b.vz = -b.vz * 0.52; b.vx *= 0.82; b.vy *= 0.82; } else b.vz = 0; }
    }
    const sp = hyp(b.vx, b.vy);
    if (sp > 0) { const dec = b.z > 0.02 ? 0.06 * sp : 2.0 + 0.1 * sp, f = Math.max(0, sp - dec * h) / sp; b.vx *= f; b.vy *= f; }
    if (sp < 0.05 && b.z === 0) { b.vx = b.vy = 0; }
    if (bounds(ox, oy)) break;
  }
  b.rot += hyp(b.vx, b.vy) * dt * 1.4;
}
function bounds(ox, oy) {
  const b = G.ball;
  if (G.phase === 'goal') {
    if (b.x < 0 || b.x > PW) {
      const lim = b.x < 0 ? -GD + 0.25 : PW + GD - 0.25;
      if ((b.x < lim && b.x < 0) || (b.x > lim && b.x > PW)) { b.x = lim; b.vx *= -0.15; Snd.net(); }
      if (b.y < GY0 + 0.15) { b.y = GY0 + 0.15; b.vy *= -0.3; } if (b.y > GY1 - 0.15) { b.y = GY1 - 0.15; b.vy *= -0.3; }
      b.vx *= 0.9; b.vy *= 0.9; if (b.z > GH) { b.z = GH; b.vz = 0; }
    }
    return false;
  }
  const pens = G.phase === 'pens';
  if (G.phase !== 'play' && !(pens && G.pens.state === 'fly')) return false;
  if ((ox >= 0 && b.x < 0) || (ox <= PW && b.x > PW)) {
    const left = b.x < 0, f = left ? ox / (ox - b.x) : (PW - ox) / (b.x - ox), yc = oy + (b.y - oy) * f, zc = b.z;
    if (zc < GH + 0.15) {
      for (const py of [GY0, GY1]) {
        if (Math.abs(yc - py) < 0.2) {
          b.x = left ? 0.15 : PW - 0.15; b.vx = -b.vx * 0.55; b.vy = (yc > py ? 1 : -1) * rnd(1, 5) + b.vy * 0.4;
          Snd.post(); G.shake = Math.max(G.shake, 0.5); emit(10, b.x, yc, { s0: 2, s1: 7, col: '#ffffff', l0: .2, l1: .5, z0: .08, z1: .14 });
          return false;
        }
      }
    }
    if (yc > GY0 && yc < GY1) {
      if (zc > GH - 0.15 && zc < GH + 0.2) {
        b.x = left ? 0.15 : PW - 0.15; b.vx = -b.vx * 0.5; b.vz = -Math.abs(b.vz) * 0.5 - 1.5;
        Snd.post(); G.shake = Math.max(G.shake, 0.5); emit(10, b.x, yc, { s0: 2, s1: 7, col: '#ffffff', l0: .2, l1: .5, z0: .08, z1: .14 });
        return false;
      }
      if (zc < GH - 0.15) { scoreGoal(left); return true; }
    }
    return goalLineOut(left);
  }
  if (b.y < 0 || b.y > PH) {
    if (pens) { pensResult('miss'); return true; }
    const team = 1 - b.lastT;
    setRestart('throw', team, clamp(b.x, 3, PW - 3), b.y < 0 ? 0 : PH);
    return true;
  }
  return false;
}
function goalLineOut(left) {
  if (G.phase === 'pens') { pensResult('miss'); return true; }
  const b = G.ball, defT = left ? G.teams.find(t => t.dir > 0) : G.teams.find(t => t.dir < 0), attT = G.teams[1 - defT.i];
  if (b.lastT === defT.i) setRestart('corner', attT.i, left ? 0.3 : PW - 0.3, b.y < CY ? 0.3 : PH - 0.3);
  else setRestart('goalkick', defT.i, left ? 5.5 : PW - 5.5, CY);
  return true;
}

function contacts() {
  const b = G.ball; if (b.owner) return;
  if (G.phase !== 'play' && G.phase !== 'pens') return;
  const sp = hyp(b.vx, b.vy);
  for (const gk of G.all) {
    if (gk.out || gk.role !== 'GK' || gk.cool > 0) continue;
    if (b.lastT === gk.team && sp > 4) continue;
    const rad = gk.diveT > 0 ? 2.1 : 1.35, d = hyp(b.x - gk.x, b.y - gk.y);
    if (d > rad || b.z > 2.6) continue;
    if (gk.diveT > 0 && !gk.saveOK) continue;
    const shot = b.shot >= 0;
    if (G.phase === 'pens') G.pens.saved = true;
    if (sp > 7) {
      if (Math.random() < clamp(0.75 - sp / 55 + gk.r / 400 - (gk.diveT > 0 ? 0.15 : 0), 0.15, 0.9)) {
        if (shot) { G.teams[gk.team].stats.saves++; banner('¡PARADA!', gk.name, G.teams[gk.team].kit.main, 1); }
        setOwner(gk); Snd.save(); return;
      }
      b.vx = -b.vx * 0.35 + rnd(-2, 2); b.vy = b.vy * 0.2 + rnd(-7, 7); b.vz = rnd(1, 4);
      if (shot) { G.teams[gk.team].stats.saves++; banner('¡PARADÓN!', gk.name, G.teams[gk.team].kit.main, 1); }
      b.shot = -1; b.last = gk; b.lastT = gk.team; b.passTeam = -1; gk.cool = 0.5; Snd.save(); G.shake = Math.max(G.shake, 0.3); return;
    }
    setOwner(gk); return;
  }
  if (G.phase === 'pens') return;
  let best = null, bd = 9;
  for (const p of G.all) {
    if (p.out || p.role === 'GK' || p.cool > 0 || p.stun > 0 || p.diveT > 0 || b.z > 2.0) continue;
    const rad = p.slideT > 0 ? 1.7 : p.tackleT > 0 ? 1.4 : 1.05, d = hyp(b.x - p.x, b.y - p.y);
    if (d > rad) continue;
    let pc = clamp(1.15 - sp / 24, 0.18, 1) * (0.72 + p.r / 330);
    if (b.target === p) pc += 0.25; if (p.ctl) pc += 0.08; if (b.z > 1) pc *= 0.75;
    if (Math.random() < pc * 0.45 && d < bd) { best = p; bd = d; }
  }
  if (best) setOwner(best);
}

function humanInput(c) {
  const m = c.map; let mx = 0, my = 0;
  if (anyDown(m.l)) mx -= 1; if (anyDown(m.r)) mx += 1; if (anyDown(m.u)) my -= 1; if (anyDown(m.d)) my += 1;
  const l = hyp(mx, my); if (l > 1) { mx /= l; my /= l; }
  return { mx, my, sprint: anyDown(m.sprint) };
}
function nearestToBall(T, exclude) {
  const b = G.ball; let best = null, bd = 1e9;
  for (const p of outfield(T)) {
    if (p === exclude) continue;
    const d = hyp(p.x - b.x, p.y - b.y); if (d < bd) { bd = d; best = p; }
  }
  return best;
}
function updateHumans(dt) {
  const b = G.ball;
  for (const c of G.hum) {
    const T = G.teams[c.team]; let cur = c.player; const R = G.restart;
    if (cur && cur.out) cur = null;
    if (G.phase === 'restart' && R.team === c.team && R.taker.role !== 'GK') cur = R.taker;
    else if (b.owner && b.owner.team === c.team && b.owner.role !== 'GK') cur = b.owner;
    else if (!b.owner && b.target && b.target.team === c.team && b.target.role !== 'GK') cur = b.target;
    else {
      c.lock -= dt;
      const best = nearestToBall(T, null);
      if (!cur || cur.role === 'GK' || (c.lock <= 0 && best && hyp(best.x - b.x, best.y - b.y) < hyp(cur.x - b.x, cur.y - b.y) - 1.5)) cur = best;
    }
    c.player = cur;
  }
}
function switchPlayer(c) {
  const T = G.teams[c.team], b = G.ball, cur = c.player;
  const list = outfield(T).filter(p => p !== cur).sort((a, d) => hyp(a.x - b.x, a.y - b.y) - hyp(d.x - b.x, d.y - b.y));
  if (list[0]) { c.player = list[0]; c.lock = 1.3; Snd.click(); }
}
function humanActions(c, dt) {
  const p = c.player; if (!p || p.out) return;
  const m = c.map, b = G.ball, inp = humanInput(c), T = G.teams[p.team], d = T.dir;
  const taker = G.phase === 'restart' && G.restart.taker === p, has = b.owner === p;
  p.ctl = c;
  if (taker) { if (inp.mx || inp.my) p.face = Math.atan2(inp.my, inp.mx); p.wx = p.wy = 0; }
  else { p.wx = inp.mx; p.wy = inp.my; p.sprint = inp.sprint; }
  let ax = inp.mx, ay = inp.my; if (!ax && !ay) { ax = Math.cos(p.face); ay = Math.sin(p.face); } else { const l = hyp(ax, ay); ax /= l; ay /= l; }
  for (const a of ['pass', 'shoot', 'lob', 'through']) {
    const dn = anyDown(m[a]), up = anyEdge(edge.up, m[a]), pr = anyEdge(edge.dn, m[a]);
    if (has && (G.phase === 'play' || taker)) {
      if (dn) c.charge[a] = Math.min(1.2, c.charge[a] + dt);
      if (up && (c.charge[a] > 0 || pr)) { fireAction(c, p, a, c.charge[a], ax, ay, inp); c.charge.pass = c.charge.shoot = c.charge.lob = c.charge.through = 0; break; }
    } else { c.charge[a] = 0; if (a === 'pass' && pr && G.phase === 'play') switchPlayer(c); }
  }
  if (anyEdge(edge.dn, m.sw) && G.phase === 'play') switchPlayer(c);
  if (anyEdge(edge.dn, m.tackle) && !has && G.phase === 'play') {
    const o = b.owner; if (o && o.team !== p.team) p.face = Math.atan2(o.y - p.y, o.x - p.x);
    startTackle(p, inp.sprint);
  }
}
function fireAction(c, p, a, ch, ax, ay, inp) {
  const d = G.teams[p.team].dir;
  if (a === 'shoot') {
    const aimY = inp.my !== 0 ? inp.my * 0.85 : (Math.random() < 0.5 ? -1 : 1) * rnd(0.15, 0.5);
    doShoot(p, aimY, 0.35 + 0.65 * clamp(ch / 0.85, 0, 1)); return;
  }
  const power = clamp(ch / 0.7, 0, 1);
  if (a === 'pass') {
    const t = bestTarget(p, ax, ay, 0.4);
    if (t) doPass(p, t.m, 'ground', power); else kickBall(p, ax, ay, 12 + power * 8, 0, {});
  } else if (a === 'lob') {
    const t = bestTarget(p, ax, ay, 0.3);
    if (t) doPass(p, t.m, 'lob', power); else lobTo(p, p.x + ax * 26, p.y + ay * 16);
  } else if (a === 'through') {
    const t = bestTarget(p, ax, ay, 0.35, m => (m.x - p.x) * d * 0.6);
    if (t) doPass(p, t.m, 'through', power); else kickBall(p, ax, ay, 16 + power * 6, 0, {});
  }
}

function pensResult(kind) {
  const P = G.pens; if (P.state === 'result') return;
  P.state = 'result'; P.t = 0;

function pensLogic(dt) {
  const P = G.pens; P.t += dt;

  if (P.state === 'aim') {
    if (c2) { const i2 = humanInput(c2); P.gkChoice = Math.abs(i2.my) > 0.4 ? Math.sign(i2.my) : 0; }
    if (c) {
      const inp = humanInput(c); P.aim = clamp(P.aim + inp.my * dt * 1.8, -1, 1);
      const keysCh = [...c.map.shoot, ...c.map.pass];
      if (anyDown(keysCh)) P.chg = Math.min(1.2, P.chg + dt);
      if ((anyEdge(edge.up, keysCh) && P.chg > 0) || P.t > 14) pensKick(0.35 + 0.65 * clamp(P.chg / 0.9, 0, 1));
      else if (anyEdge(edge.up, keysCh)) pensKick(0.5);
    } else if (P.t > 1.8) { P.aim = pick([-1, 1]) * rnd(0.3, 0.9); pensKick(rnd(0.6, 0.95)); }

}

function applySnap(a) {
  const b = G.ball; b.x = a[0]; b.y = a[1]; b.z = a[2]; b.rot = a[3]; b.owner = null; b.vx = b.vy = 0;

function replayStep(dt) {
  const R = G.replay; R.i += dt * 30 * 0.6;

function camUpdate(dt) {
  const c = G.cam, b = G.ball;
  let tx = b.x + clamp(b.vx * 0.3, -8, 8), ty = b.y + clamp(b.vy * 0.3, -4, 4), vw = 46;
  if (b.owner) { tx = b.owner.x + Math.cos(b.owner.face) * 4; ty = b.owner.y + Math.sin(b.owner.face) * 1.5; }
  if (G.phase === 'pens') { tx = PW - 13; ty = CY; vw = 30; }
  else if (G.phase === 'goal') { tx = b.x; ty = b.y; vw = 42; }
  else if (G.phase === 'restart' && G.restart.type === 'pen') { tx = G.restart.goalX > PW / 2 ? PW - 14 : 14; ty = CY; vw = 34; }
  c.vw += (vw - c.vw) * (1 - Math.exp(-dt * 2));
  const k = 1 - Math.exp(-dt * (G.phase === 'replay' ? 3 : 4));
  c.x = clamp(c.x + (tx - c.x) * k, -4, PW + 4); c.y = clamp(c.y + (ty - c.y) * k, -2, PH + 2);
}
function partsUpdate(dt) {
  for (let i = G.parts.length - 1; i >= 0; i--) {
    const q = G.parts[i]; q.life -= dt;
    if (q.life <= 0) { G.parts.splice(i, 1); continue; }
    q.x += q.vx * dt; q.y += q.vy * dt; q.z += q.vz * dt; q.vz -= q.g * dt; q.rot += q.vr * dt;
    if (q.drag) { const f = Math.exp(-q.drag * dt); q.vx *= f; q.vy *= f; }
    if (q.z < 0) { q.z = 0; q.vz *= -0.3; q.vx *= 0.6; q.vy *= 0.6; }
  }
}
function update(dt) {
  G.t += dt;
  if (G.shake > 0) G.shake = Math.max(0, G.shake - dt * 2.2);
  if (G.flash > 0) G.flash = Math.max(0, G.flash - dt * 1.6);
  if (G.banner) { G.banner.t += dt; if (G.banner.t > G.banner.dur) G.banner = null; }
  if (G.card) { G.card.t += dt; if (G.card.t > 2.4) G.card = null; }
  for (const q of G.ticker) q.t += dt;
  while (G.ticker.length && G.ticker[G.ticker.length - 1].t > 9) G.ticker.pop();
  const ph = G.phase;
  for (const p of G.all) {
    if (p.cool > 0) p.cool -= dt; if (p.stun > 0) p.stun -= dt; if (p.tackleT > 0) p.tackleT -= dt;
    if (p.slideT > 0) p.slideT -= dt; if (p.diveT > 0) p.diveT -= dt; if (p.tkCool > 0) p.tkCool -= dt;
    if (p.diveCool > 0) p.diveCool -= dt; if (p.kick > 0) p.kick -= dt; p.ctl = null;
  }
  if (ph === 'replay') { replayStep(dt); partsUpdate(dt); camUpdate(dt); return; }
  if (ph === 'play') { G.clock += dt; if (G.ball.owner) G.teams[G.ball.owner.team].stats.poss += dt; }
  if (ph === 'play' || ph === 'restart') { updateHumans(dt); for (const c of G.hum) humanActions(c, dt); }
  computeRanks();
  for (const p of G.all) if (!p.ctl) aiControl(p);
  if (ph === 'play') resolveTackles();
  for (const p of G.all) movePlayer(p, dt);
  separation();
  ballUpdate(dt);
  contacts();
  const ph2 = G.phase;
  if (ph2 === 'restart') {
    const R = G.restart; R.t += dt;
    if (!R.taker.ctl && R.t > R.wait) aiRestartKick(); else if (R.t > 15) aiRestartKick();
  } else if (ph2 === 'play') {
    G.recT++; if (G.recT % 2 === 0) record();
    if (G.half === 1 && G.clock >= G.len / 2) { G.phase = 'half'; G.timer = 3; banner('DESCANSO', `${G.teams[0].def.s} ${G.teams[0].score} - ${G.teams[1].score} ${G.teams[1].def.s}`, '#ffffff', 2.8); Snd.whistle(true); }
    else if (G.clock >= G.len && !(G.ball.shot >= 0 && hyp(G.ball.vx, G.ball.vy) > 8)) {
      G.phase = 'end'; G.timer = 2.4; banner('FINAL DEL PARTIDO', `${G.teams[0].def.s} ${G.teams[0].score} - ${G.teams[1].score} ${G.teams[1].def.s}`, '#ffffff', 2.4); Snd.whistle(true);
    }
  } else if (ph2 === 'half') {
    G.timer -= dt;
    if (G.timer <= 0) { G.half = 2; G.teams.forEach(t => { t.dir = -t.dir; }); setKickoff(1); }
  } else if (ph2 === 'goal') {
    G.timer -= dt;
    if (G.timer <= 0) { if (G.o.replay && !G.demo) startReplay(); else afterGoal(); }
  } else if (ph2 === 'end') {
    G.timer -= dt;
    if (G.timer <= 0 && !G.ended) {
      if (G.demo) { G.ended = true; return; }
      if (!G.penEnd && G.teams[0].score === G.teams[1].score && G.o.pens) startShootout();
      else { G.ended = true; if (!G.penEnd) G.winner = G.teams[0].score > G.teams[1].score ? 0 : G.teams[0].score < G.teams[1].score ? 1 : -1; S.onEnd && S.onEnd(); }
    }
  } else if (ph2 === 'pens') pensLogic(dt);
  if (G.penEnd && G.phase === 'end' && G.timer <= 0 && !G.ended && !G.demo) { G.ended = true; S.onEnd && S.onEnd(); }
  camUpdate(dt); partsUpdate(dt);
}

function startDemo() {
  const a = Math.floor(Math.random() * TEAMS.length); let b; do { b = Math.floor(Math.random() * TEAMS.length); } while (b === a);
  newMatch({ t: [a, b], f: [Math.floor(Math.random() * FORMS.length), Math.floor(Math.random() * FORMS.length)], len: 1e9, diff: 1, pens: false, replay: false, demo: true, c: [null, null] });
  G.diff = 2; Snd.quiet = true;
}
function startGame() {
  const m = S.mode, o = { t: [S.sel[0], S.sel[1]], f: [S.opt.f[0], S.opt.f[1]], len: LENS[S.opt.len].s, diff: S.opt.diff, pens: S.opt.pens, replay: S.opt.replay,
    c: m === '1p' ? [MAP1, null] : m === '2p' ? [MAP2A, MAP2B] : [null, null] };
  Snd.init(); Snd.resume(); Snd.on = S.opt.sound; Snd.quiet = false;
  newMatch(o); S.screen = 'play'; S.hint = 16;
  if (S.pensOnly) { G.o.pens = true; G.clock = G.len; G.half = 2; startShootout(); } keys.clear(); edge.dn.clear(); edge.up.clear();
  S.onEnd = () => { S.screen = 'end'; S.endRow = 0; Snd.crowd(0.1, 0.2); };
}
const inside = (mx, my, h) => mx >= h.x && mx <= h.x + h.w && my >= h.y && my <= h.y + h.h;
function handleUI() {
  const dn = edge.dn, has = c => dn.has(c);
  const up = has('ArrowUp') || has('KeyW'), down = has('ArrowDown') || has('KeyS'), left = has('ArrowLeft') || has('KeyA'), right = has('ArrowRight') || has('KeyD');
  const ok = has('Enter') || has('Space') || has('PadA'), back = has('Escape') || has('Backspace') || has('PadB');
  const click = mouse.click; mouse.click = false;
  const sc = S.screen;
  if (sc === 'play') {
    if (has('Escape') || has('KeyP')) { S.screen = 'pause'; S.pauseRow = 0; dn.clear(); edge.up.clear(); keys.clear(); return; }
    if (G && G.phase === 'replay' && (has('Enter') || has('Space'))) { G.replay.skip = true; dn.delete('Enter'); dn.delete('Space'); }
    return;
  }
  if (S.hitsPrev) {
    if (mouse.moved) for (const h of S.hitsPrev) if (h.hover && inside(mouse.x, mouse.y, h)) h.hover();
    if (click) for (let i = S.hitsPrev.length - 1; i >= 0; i--) { const h = S.hitsPrev[i]; if (inside(mouse.x, mouse.y, h)) { h.fn(); break; } }
  }
  mouse.moved = false;
  if (S.screen === sc) {
    if (sc === 'title') { if (ok) { Snd.init(); Snd.resume(); S.screen = 'mode'; S.mi = 0; S.si = -1; } }
    else if (sc === 'mode') { menuNav(up, down); if (ok) menuAction(); if (back) { if (S.si >= 0 && MENU[S.mi].subs) S.si = -1; else S.screen = 'title'; } }
    else if (sc === 'teams') {
      const n = TEAMS.length; let moved = true;
      if (left) S.cur = (S.cur + n - 1) % n; else if (right) S.cur = (S.cur + 1) % n; else if (up) S.cur = (S.cur + n - 4) % n; else if (down) S.cur = (S.cur + 4) % n; else moved = false;
      if (moved) Snd.click();
      if (has('Enter') || has('PadA')) teamsOk();
      if (back) { if (S.step === 1) { S.step = 0; S.cur = S.sel[0]; } else S.screen = 'mode'; }
    } else if (sc === 'opts') {
      const n = optRows().length;
      if (up) { S.row = (S.row + n - 1) % n; Snd.click(); } if (down) { S.row = (S.row + 1) % n; Snd.click(); }
      if (left) optChange(S.row, -1); if (right) optChange(S.row, 1);
      if (ok) { const r = optRows()[S.row]; if (r.k === 'go') startGame(); else optChange(S.row, 1); }
      if (back) S.screen = 'teams';
    } else if (sc === 'help') { if (ok || back) S.screen = S.helpBack; }
    else if (sc === 'pause') {
      if (up) { S.pauseRow = (S.pauseRow + 3) % 4; Snd.click(); } if (down) { S.pauseRow = (S.pauseRow + 1) % 4; Snd.click(); }
      if (ok) pauseOk(); if (back || has('KeyP')) S.screen = 'play';
    } else if (sc === 'end') {
      if (left || right || up || down) { S.endRow = 1 - S.endRow; Snd.click(); } if (ok) endOk();
    }
  }
  dn.clear(); edge.up.clear();
}
function render() {
  ctx.setTransform(dpr, 0, 0, dpr, 0, 0); ctx.fillStyle = '#05080f'; ctx.fillRect(0, 0, W, H);
  S.hits = [];
  const sc = S.screen;
  if (sc === 'play') { drawWorld(); drawHUD(); }
  else if (sc === 'title') drawTitle(); else if (sc === 'mode') drawMode(); else if (sc === 'teams') drawTeams(); else if (sc === 'opts') drawOpts();
  else if (sc === 'help') drawHelp(); else if (sc === 'pause') { drawWorld(); drawHUD(); drawPause(); } else if (sc === 'end') drawEnd();
  S.hitsPrev = S.hits;
  cv.style.cursor = S.hits.some(h => inside(mouse.x, mouse.y, h)) ? 'pointer' : 'default';
}
function crowdAudio(dt) {
  if (!G || !Snd.ctx) return;
  const b = G.ball, dg = Math.min(hyp(b.x, b.y - CY), hyp(PW - b.x, b.y - CY)), danger = 1 - clamp(dg / 38, 0, 1);
  Snd.boost = Math.max(0, Snd.boost - dt * 0.35);
  const base = Snd.quiet ? 0 : 0.1 + danger * 0.16 + Snd.boost * 0.5;
  Snd.crowd(S.screen === 'pause' ? base * 0.3 : base, 0.15 + danger * 0.6 + Snd.boost * 0.3);
}
let acc = 0;
function frame(ts) {
  raf = requestAnimationFrame(frame);
  if (!S.last) S.last = ts;
  const dt = Math.min(0.05, (ts - S.last) / 1000); S.last = ts; S.t += dt;
  if (S.hint > 0 && S.screen === 'play') S.hint -= dt;
  pollPad();
  handleUI();
  if (G && S.screen !== 'pause' && S.screen !== 'end' && !(S.screen === 'help' && S.helpBack === 'pause')) {
    acc += dt; let n = 0;
    while (acc >= DT && n++ < 5) { update(DT); edge.dn.clear(); edge.up.clear(); acc -= DT; }
    if (G.demo && G.ended) startDemo();
    if (G.demo && G.phase === 'end') G.phase = 'play';
  }
  crowdAudio(dt);
  render();
}

const onKeyDown = e => {
  if (e.ctrlKey || e.metaKey || e.altKey || /^F\d+$/.test(e.code)) return;
  e.preventDefault(); e.stopImmediatePropagation();
  Snd.resume();
  if (!e.repeat) { keys.add(e.code); edge.dn.add(e.code); }
  else if (S.screen !== 'play') edge.dn.add(e.code);
};
const onKeyUp = e => { if (e.ctrlKey || e.metaKey || e.altKey) return; e.preventDefault(); e.stopImmediatePropagation(); keys.delete(e.code); edge.up.add(e.code); };
const onBlur = () => { keys.clear(); };
const onMove = e => { const r = cv.getBoundingClientRect(); mouse.x = e.clientX - r.left; mouse.y = e.clientY - r.top; mouse.moved = true; };
const onDown = e => { onMove(e); mouse.click = true; cv.focus(); Snd.init(); Snd.resume(); };
window.addEventListener('keydown', onKeyDown, true); window.addEventListener('keyup', onKeyUp, true);
window.addEventListener('blur', onBlur); window.addEventListener('resize', resize);
cv.addEventListener('mousemove', onMove); cv.addEventListener('mousedown', onDown);
cv.addEventListener('contextmenu', e => e.preventDefault());

  cv.remove(); delete window.FC27;
  console.log('%cFC 27 cerrado. Pega el script otra vez para volver a jugar.', 'color:#22d3ee;font-weight:bold');
}
